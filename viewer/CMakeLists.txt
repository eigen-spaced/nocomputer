cmake_minimum_required(VERSION 3.24)
project(NoComputerViewerPoC LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(SDL3 CONFIG REQUIRED)
if(WIN32)
  find_package(FFMPEG REQUIRED)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavformat libavutil libswscale)
endif()
add_executable(nocomputer-viewer src/main.cpp)
target_link_libraries(nocomputer-viewer PRIVATE SDL3::SDL3)
if(WIN32)
  target_compile_definitions(nocomputer-viewer PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(nocomputer-viewer PRIVATE ${FFMPEG_INCLUDE_DIRS})
  target_link_directories(nocomputer-viewer PRIVATE ${FFMPEG_LIBRARY_DIRS})
  target_link_libraries(nocomputer-viewer PRIVATE ${FFMPEG_LIBRARIES})
else()
  target_link_libraries(nocomputer-viewer PRIVATE PkgConfig::FFMPEG)
endif()
if(MSVC)
  target_compile_options(nocomputer-viewer PRIVATE /W4 /permissive-)
else()
  target_compile_options(nocomputer-viewer PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
add_executable(viewer-geometry-tests tests/geometry.cpp)
add_test(NAME geometry COMMAND viewer-geometry-tests)
add_test(NAME rejects-network-url COMMAND nocomputer-viewer --check https://example.invalid/fixture.h264)
set_tests_properties(rejects-network-url PROPERTIES WILL_FAIL TRUE)
add_test(NAME rejects-malformed-h264 COMMAND nocomputer-viewer --software --check ${CMAKE_CURRENT_SOURCE_DIR}/tests/invalid.h264)
set_tests_properties(rejects-malformed-h264 PROPERTIES WILL_FAIL TRUE)
set(NOCOMPUTER_FIXTURE_PATH "" CACHE FILEPATH "Generated synthetic H.264 fixture for offline integration tests")
if(NOCOMPUTER_FIXTURE_PATH)
  add_test(NAME software-decode COMMAND nocomputer-viewer --software --check ${NOCOMPUTER_FIXTURE_PATH})
  set_tests_properties(software-decode PROPERTIES PASS_REGULAR_EXPRESSION "\"frames\":30,\"hardwareFrames\":0" TIMEOUT 30)
  add_test(NAME texture-upload COMMAND nocomputer-viewer --software --smoke ${NOCOMPUTER_FIXTURE_PATH})
  set_tests_properties(texture-upload PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software"
    PASS_REGULAR_EXPRESSION "\"frames\":30,\"hardwareFrames\":0" TIMEOUT 30)
  if(NOT WIN32)
    add_test(NAME rejects-windows-hardware-on-other-platform COMMAND nocomputer-viewer --require-hardware --check ${NOCOMPUTER_FIXTURE_PATH})
    set_tests_properties(rejects-windows-hardware-on-other-platform PROPERTIES WILL_FAIL TRUE TIMEOUT 30)
  endif()
endif()
